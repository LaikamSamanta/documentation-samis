JavaScript

js(function () {
    const car = document.getElementById('dscar1');
    const track = car.querySelector('.dscar-track');
    const slides = car.querySelectorAll('.dscar-slide');
    const dotsWrap = car.querySelector('.dscar-dots');
    let index = 0;

    slides.forEach((_, i) => {
        const dot = document.createElement('button');
        if (i === 0) dot.classList.add('active');
        dot.addEventListener('click', () => goTo(i));
        dotsWrap.appendChild(dot);
    });
    const dots = dotsWrap.querySelectorAll('button');

    function goTo(i) {
        index = (i + slides.length) % slides.length;
        track.style.transform = `translateX(-${index * 100}%)`;
        dots.forEach((d, di) => d.classList.toggle('active', di === index));
    }

    car.querySelector('.dscar-prev').addEventListener('click', () => goTo(index - 1));
    car.querySelector('.dscar-next').addEventListener('click', () => goTo(index + 1));

    let timer = setInterval(() => goTo(index + 1), 4000);
    car.addEventListener('mouseenter', () => clearInterval(timer));
    car.addEventListener('mouseleave', () => { timer = setInterval(() => goTo(index + 1), 4000); });
})();

Attēli šeit ir reāli Unsplash foto — nomaini src uz savām bildēm. Ja lapā ir vairāki karuseļi, katram vajag unikālu id un savu document.getElementById(...) izsaukumu. Pirmajam slaidam der ielikt loading="eager", pārējiem loading="lazy", lai neaizkavētu LCP (skat. SEO → Attēlu optimizācija).

JavaScript

jsdocument.querySelectorAll('.dsacc-item').forEach((item) => {
    const head = item.querySelector('.dsacc-head');
    const body = item.querySelector('.dsacc-body');

    head.addEventListener('click', () => {
        const isOpen = item.classList.contains('open');

        item.parentElement.querySelectorAll('.dsacc-item').forEach((i) => {
            i.classList.remove('open');
            i.querySelector('.dsacc-body').style.maxHeight = null;
        });

        if (!isOpen) {
            item.classList.add('open');
            body.style.maxHeight = body.scrollHeight + 'px';
        }
    });
});

JavaScript

jsdocument.querySelectorAll('.dstabs').forEach((tabs) => {
    const nav = tabs.querySelector('.dstabs-nav');
    const indicator = document.createElement('div');
    indicator.className = 'dstabs-indicator';
    nav.prepend(indicator);
    const btns = tabs.querySelectorAll('.dstabs-btn');

    function moveIndicator(btn) {
        indicator.style.width = btn.offsetWidth + 'px';
        indicator.style.transform = `translateX(${btn.offsetLeft}px)`;
    }

    btns.forEach((btn) => {
        btn.addEventListener('click', () => {
            btns.forEach((b) => b.classList.remove('active'));
            tabs.querySelectorAll('.dstabs-panel').forEach((p) => p.classList.remove('active'));
            btn.classList.add('active');
            tabs.querySelector('#' + btn.dataset.tab).classList.add('active');
            moveIndicator(btn);
        });
    });

    moveIndicator(tabs.querySelector('.dstabs-btn.active'));
});

moveIndicator() mēra pogas offsetWidth/offsetLeft — tas strādā pareizi tikai pēc tam, kad pogas jau ir reāli renderētas lapā (tāpēc pēdējā rinda izsauc to uzreiz pēc ielādes, lai indikators uzreiz stāv zem aktīvās pogas, nevis lec turp pie pirmā klikšķa).

JavaScript

jsconst thumbs = Array.from(document.querySelectorAll('.dslb-thumb'));
const overlay = document.getElementById('lbOverlay');
const lbImg = document.getElementById('lbImage');
let current = 0;

function openAt(i) {
    current = i;
    lbImg.src = thumbs[i].dataset.full || thumbs[i].src;
    lbImg.alt = thumbs[i].alt;
    overlay.classList.add('open');
}
function next() { openAt((current + 1) % thumbs.length); }
function prev() { openAt((current - 1 + thumbs.length) % thumbs.length); }

thumbs.forEach((thumb, i) => thumb.addEventListener('click', () => openAt(i)));
document.getElementById('lbClose').addEventListener('click', () => overlay.classList.remove('open'));
document.getElementById('lbNext').addEventListener('click', next);
document.getElementById('lbPrev').addEventListener('click', prev);
overlay.addEventListener('click', (e) => {
    if (e.target === overlay) overlay.classList.remove('open'); // ne uz attēla/pogām
});
document.addEventListener('keydown', (e) => {
    if (!overlay.classList.contains('open')) return;
    if (e.key === 'Escape') overlay.classList.remove('open');
    if (e.key === 'ArrowRight') next();
    if (e.key === 'ArrowLeft') prev();
});

data-full ļauj sīktēlam ielādēt mazu attēlu (ātri, lēti), bet lightbox skatam parādīt lielu versiju — reālā projektā ģenerē abus izmērus servera pusē (skat. SEO → Attēlu optimizācija). Ja data-full nav norādīts, kods vienkārši lieto to pašu src, kas jau ir — demonstrācijā abi sakrīt.
